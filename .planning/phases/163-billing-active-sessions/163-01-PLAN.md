```typescript
export interface ActiveSession {
  id: string;
  driver_id: string;
  driver_name: string;
  pod_id: string;
  pod_number: number;
  pricing_tier_name: string;
  allocated_seconds: number;

  remaining_seconds: number;
  status: 'active' | 'paused_manual' | 'completed' | 'cancelled' | 'expired';
  price_paise: number;
  started_at: string;
  paused_at: string | null;
  ended_at: string | null;
  staff_id: string | null;
  staff_name: string | null;
}

export interface SessionEvent {
  id: string;
  session_id: string;
  event_type: string;   // 'start' | 'pause' | 'resume' | 'extend' | 'stop'
  timestamp: string;
  details: string | null;
  staff_name: string | null;
}

export interface BillingRate {
  id: string;
  name: string;
  duration_minutes: number;
  price_paise: number;
  active: boolean;
}
```

Build billingApi object with these methods:
- `getActive(): Promise<ActiveSession[]>` -> GET `/billing/active`
- `getSession(id: string): Promise<ActiveSession>` -> GET `/billing/sessions/${id}`
- `startSession(data: { pod_id: string; driver_id?: string; rate_id: string }): Promise<ActiveSession>` -> POST `/billing/start` with JSON body
- `stopSession(id: string): Promise<void>` -> POST `/billing/${id}/stop`
- `pauseSession(id: string): Promise<void>` -> POST `/billing/${id}/pause`
- `resumeSession(id: string): Promise<void>` -> POST `/billing/${id}/resume`
- `extendSession(id: string, minutes: number): Promise<void>` -> POST `/billing/${id}/extend` with JSON body `{ minutes }`
- `getSessionEvents(id: string): Promise<SessionEvent[]>` -> GET `/billing/sessions/${id}/events`
- `getRates(): Promise<BillingRate[]>` -> GET `/billing/rates`

All methods use `rcFetch` from `./base`. POST methods pass `{ method: 'POST', body: JSON.stringify(payload) }`.

**Create `src/app/(dashboard)/billing/page.tsx`:**

A 'use client' page component that:

1. Uses `useSWR<ActiveSession[]>('/billing/active', () => billingApi.getActive(), { refreshInterval: 5000 })` for live data.

2. Has a `useEffect` with a 1-second `setInterval` that decrements a local `countdowns` state (Record<string, number>) derived from `data`. On each SWR refresh, re-sync countdowns from server `remaining_seconds`. This gives smooth per-second countdown between 5s polls.

3. Renders a header row: "Active Sessions" title on left, session count badge, and a "Start Session" button (disabled for now — wired in Plan 02) on right.

4. Table columns: Pod | Customer | Rate/Tier | Time Remaining (countdown) | Elapsed | Status | Price | Actions
   - Pod: `pod_number` displayed as "Pod N"
   - Customer: `driver_name`
   - Rate: `pricing_tier_name`
   - Time Remaining: formatted as MM:SS or H:MM:SS from local countdown state. Color-coded:
     - `text-emerald-400` when countdown > 1800s (30 min)
     - `text-yellow-400` when countdown <= 1800s and > 300s
     - `text-red-400 font-bold` when countdown <= 300s (5 min) — also add `animate-pulse` class
   - Elapsed: `driving_seconds` formatted as Xh Ym or Ym
   - Status: badge using same `statusBadge()` pattern from sessions page (green for active, yellow for paused_manual)
   - Price: formatted as Indian rupees from `price_paise` (divide by 100, use INR locale)
   - Actions: placeholder column with empty cells (buttons added in Plan 02)

5. Table styling matches fleet page pattern: `bg-rp-card border border-rp-border rounded-xl overflow-hidden`, `w-full text-sm`, header row with `border-b border-rp-border text-rp-grey text-left`, body rows with `border-b border-rp-border/50 last:border-0 hover:bg-white/[0.02]`.

6. Empty state: when no active sessions, show centered message "No active sessions" in a card with a prominent "Start Session" button.

7. Loading state: show "Loading sessions..." centered text (same pattern as fleet).

8. Error state: red error banner with retry button (same pattern as fleet page).
